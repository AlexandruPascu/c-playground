cmake_minimum_required(VERSION 3.20)
project(CPlayground LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PLAYGROUND_BUILD_GRAPHICS "Build the Allegro windowed examples" OFF)
option(PLAYGROUND_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(PLAYGROUND_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(PLAYGROUND_REQUIRE_INTEGRATION_TESTS "Require Python for command-line tests" OFF)

add_library(tic_tac_toe_rules PoliTicTacToe/PoliTicTacToe.c)
target_compile_definitions(tic_tac_toe_rules PRIVATE PLAYGROUND_NO_MAIN)
add_library(tetris_rules Lab05/Solved/board.c Lab05/Solved/score.c Lab05/Solved/ai.c Lab05/Solved/weights.c)
add_library(algorithms algorithms/fibonacci.c algorithms/flood_fill.c workshop/list.c workshop/pin.c workshop/image_ops.c)
add_library(image_codec workshop/stb_codec.c)
if(MSVC)
    target_compile_definitions(image_codec PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_library(workshop_apps workshop/console.c workshop/accounts.c workshop/images.c workshop/fibonacci_cli.c workshop/components_cli.c)
target_link_libraries(workshop_apps PUBLIC algorithms image_codec)
if(UNIX)
    target_link_libraries(workshop_apps PUBLIC m)
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/output")
target_compile_definitions(workshop_apps PRIVATE
    PLAYGROUND_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
    PLAYGROUND_OUTPUT_DIR="${CMAKE_CURRENT_BINARY_DIR}/output"
    PLAYGROUND_ACCOUNTS_FILE="${CMAKE_CURRENT_SOURCE_DIR}/Lab01/challenge5/challenge5.txt"
    PLAYGROUND_GRID_FILE="${CMAKE_CURRENT_SOURCE_DIR}/Lab01/challenge4/challenge4.txt")
add_executable(tic_tac_toe PoliTicTacToe/PoliTicTacToe.c)
add_executable(fibonacci Lab02/Solved/challenge1/main.c)
target_link_libraries(fibonacci PRIVATE workshop_apps)
add_executable(components Lab01/Solved/challenge4/main.c)
target_link_libraries(components PRIVATE workshop_apps)
target_compile_definitions(components PRIVATE PLAYGROUND_GRID_FILE="${CMAKE_CURRENT_SOURCE_DIR}/Lab01/Solved/challenge4/challenge4.txt")
add_executable(linked_list Lab02/Demo/demo_linkedlist.c)
add_executable(structs Lab02/Demo/demo_struct.c)
add_executable(tetris_ai Lab05/Solved/ai_cli.c)
target_link_libraries(tetris_ai PRIVATE tetris_rules)
set(playground_targets tic_tac_toe_rules tetris_rules algorithms workshop_apps tic_tac_toe fibonacci components linked_list structs tetris_ai)
set(workshop_cli_targets numbers vector_sum fps pin_lab highlight list_ops insertion_sort image_convert image_grayscale image_decode)
set(workshop_cli_sources Lab01/challenge1/main.c Lab01/challenge2/main.c Lab01/challenge3/main.c
    Lab01/challenge5/main.c Lab01/challenge6/main.c Lab02/challenge2/main.c Lab02/challenge3/main.c
    Lab02/challenge4/main.c Lab02/challenge5/main.c Lab02/challenge6/main.c)
foreach(index RANGE 0 9)
    list(GET workshop_cli_targets ${index} target)
    list(GET workshop_cli_sources ${index} source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE workshop_apps)
    list(APPEND playground_targets ${target})
endforeach()

    list(APPEND playground_targets tetris_ai_tests)

    add_executable(core_tests tests/core_tests.c)
    target_include_directories(core_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(core_tests PRIVATE tic_tac_toe_rules tetris_rules algorithms)
    add_executable(workshop_tests tests/workshop_tests.c)
    target_include_directories(workshop_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(workshop_tests PRIVATE tic_tac_toe_rules tetris_rules algorithms)
    list(APPEND playground_targets core_tests workshop_tests)
    add_test(NAME workshop_algorithms COMMAND workshop_tests)
    add_test(NAME rules_and_algorithms COMMAND core_tests)
    add_test(NAME linked_list_lifetime COMMAND linked_list)
    add_test(NAME struct_example COMMAND structs)
    if(PLAYGROUND_REQUIRE_INTEGRATION_TESTS)
        find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)

        find_package(Python3 3.8 QUIET COMPONENTS Interpreter)

    if(Python3_Interpreter_FOUND)
        add_test(NAME tetris_training COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/tetris_training_tests.py"
                 "$<TARGET_FILE:tetris_ai>" "${CMAKE_CURRENT_SOURCE_DIR}/tools/train_tetris.py")
        set_tests_properties(tetris_training PROPERTIES TIMEOUT 120)
        add_test(NAME tetris_ai_games COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/tetris_ai_cli_tests.py" "$<TARGET_FILE:tetris_ai>")

    endif()

if(PLAYGROUND_BUILD_GRAPHICS)
    include(cmake/Graphics.cmake)
endif()
if(PLAYGROUND_SANITIZERS AND (NOT UNIX OR MSVC OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang"))
    message(FATAL_ERROR "Sanitizers require GCC or Clang on a Unix platform")
endif()
foreach(target IN LISTS playground_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        # Use the portable C library; input widths and numeric ranges are checked explicitly.
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        if(PLAYGROUND_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wstrict-prototypes)
        if(PLAYGROUND_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(PLAYGROUND_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()

# Third-party codec code is instrumented too, without applying our style warnings.
if(PLAYGROUND_SANITIZERS)
    target_compile_options(image_codec PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
