cmake_minimum_required(VERSION 3.20)
project(CPlayground LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PLAYGROUND_BUILD_GRAPHICS "Build the Allegro windowed examples" OFF)
option(PLAYGROUND_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(PLAYGROUND_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(PLAYGROUND_REQUIRE_INTEGRATION_TESTS "Require Python for command-line tests" OFF)

add_library(tic_tac_toe_rules PoliTicTacToe/PoliTicTacToe.c)
target_compile_definitions(tic_tac_toe_rules PRIVATE PLAYGROUND_NO_MAIN)
add_library(tetris_rules Lab05/Solved/board.c)
add_library(algorithms algorithms/fibonacci.c algorithms/flood_fill.c)
add_executable(tic_tac_toe PoliTicTacToe/PoliTicTacToe.c)
add_executable(fibonacci Lab02/Solved/challenge1/main.c)
target_link_libraries(fibonacci PRIVATE algorithms)
add_executable(components Lab01/Solved/challenge4/main.c)
target_link_libraries(components PRIVATE algorithms)
target_compile_definitions(components PRIVATE PLAYGROUND_GRID_FILE="${CMAKE_CURRENT_SOURCE_DIR}/Lab01/Solved/challenge4/challenge4.txt")
add_executable(linked_list Lab02/Demo/demo_linkedlist.c)
add_executable(structs Lab02/Demo/demo_struct.c)
set(playground_targets tic_tac_toe_rules tetris_rules algorithms tic_tac_toe fibonacci components linked_list structs)

include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.c)
    target_include_directories(core_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(core_tests PRIVATE tic_tac_toe_rules tetris_rules algorithms)
    list(APPEND playground_targets core_tests)
    add_test(NAME rules_and_algorithms COMMAND core_tests)
    add_test(NAME linked_list_lifetime COMMAND linked_list)
    add_test(NAME struct_example COMMAND structs)
    if(PLAYGROUND_REQUIRE_INTEGRATION_TESTS)
        find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    else()
        find_package(Python3 3.8 QUIET COMPONENTS Interpreter)
    endif()
    if(Python3_Interpreter_FOUND)
        add_test(NAME command_line COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py"
                 "$<TARGET_FILE:tic_tac_toe>" "$<TARGET_FILE:fibonacci>" "$<TARGET_FILE:components>")
    else()
        message(STATUS "Python 3.8+ not found; command-line tests are unavailable")
    endif()
endif()

if(PLAYGROUND_BUILD_GRAPHICS)
    include(cmake/Graphics.cmake)
endif()
if(PLAYGROUND_SANITIZERS AND (NOT UNIX OR MSVC OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang"))
    message(FATAL_ERROR "Sanitizers require GCC or Clang on a Unix platform")
endif()
foreach(target IN LISTS playground_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        # Use the portable C library; input widths and numeric ranges are checked explicitly.
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        if(PLAYGROUND_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(PLAYGROUND_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
    if(PLAYGROUND_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
